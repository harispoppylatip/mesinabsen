# Layanan wajah untuk absen wajah aplikasi sekolah.
#
# Pemasangan ada dua langkah, dan urutannya penting:
#
#     pip install -r requirements.txt
#     pip install --no-deps face-recognition==1.3.0
#
# Langkah kedua ada di luar berkas ini dengan sengaja. Paket face-recognition
# menuliskan "dlib>=19.7" sebagai syarat, sedangkan distribusi dlib di PyPI
# hanya tersedia sebagai berkas sumber sehingga pip akan mengompilasinya
# (butuh cmake dan belasan menit). Modul dlib sudah disediakan oleh dlib-bin
# di bawah, jadi pemasangan paket itu dijalankan tanpa dependensi. Karena itu
# seluruh kebutuhan lain dari paket tersebut didaftarkan di sini.
#
# Dockerfile menjalankan kedua perintah itu dengan urutan yang sama.

# Layanan web
fastapi==0.142.3
uvicorn==0.54.0
python-multipart==0.0.32

# Mesin wajah
#
# dlib-bin menyediakan roda siap pakai untuk Windows dan Linux x86_64, jadi
# tidak ada kompilasi dari sumber. Modul yang dipasangnya bernama "dlib".
dlib-bin==20.0.1.post1

# Kebutuhan face-recognition selain dlib, karena paket itu dipasang tanpa
# dependensi.
face-recognition-models==0.3.0
numpy==1.26.4
pillow==12.1.0
click==8.3.1

# face-recognition-models masih memakai pkg_resources dari setuptools. Paket
# itu dijadwalkan dihapus, jadi versinya dikunci di bawah 81 supaya
# pemasangan tidak tiba-tiba gagal.
setuptools<81

# Dipakai skrip pendaftaran massal (enroll_faces.py).
requests==2.32.5
